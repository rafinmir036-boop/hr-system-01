const fs = require('fs'); const readline = require('readline'); const https = require('https'); const path = require('path'); const WATCH_FOLDER = 'D:/zk_logs'; console.log("🚀 ZKBio Time to HR System Sync Agent Active..."); setInterval(() => { if (!fs.existsSync(WATCH_FOLDER)) return; const files = fs.readdirSync(WATCH_FOLDER); files.forEach(file => { if (file === 'sync.js') return; const filePath = path.join(WATCH_FOLDER, file); if (file.endsWith('.txt') || file.endsWith('.csv')) { const fileStream = fs.createReadStream(filePath); const rl = readline.createInterface({ input: fileStream, crlfDelay: Infinity }); rl.on('line', (line) => { if (!line.trim()) return; const parts = line.includes('\t') ? line.split('\t') : line.split(','); if (parts.length >= 2) { const empCode = parts[0].trim(); let dateStr = parts[1] ? parts[1].trim() : new Date().toISOString().split('T')[0]; if (dateStr.includes(' ')) { dateStr = dateStr.split(' ')[0]; } if (empCode) { console.log(`📡 Sending Punch Data: Emp Code [${empCode}] on Date [${dateStr}]`); pushToFirebase(dateStr, empCode, 'Present'); } } }); } }); }, 5000); function pushToFirebase(dateStr, empId, status) { const url = `https://firestore.googleapis.com/v1/projects/rafin-328ee/databases/(default)/documents/attendance_${dateStr}/${empId}`; const payload = JSON.stringify({ fields: { employeeId: { stringValue: empId }, customEmpCode: { stringValue: empId }, status: { stringValue: status }, updatedAt: { stringValue: new Date().toISOString() } } }); const req = https.request(url, { method: 'PATCH', headers: { 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(payload) } }); req.on('response', (res) => { if (res.statusCode === 200) { console.log(`✅ Success: Attendance updated in HR System for ID [${empId}]`); } }); req.on('error', (e) => { console.error(`❌ Sync Error: ${e.message}`); }); req.write(payload); req.end(); }
